"use client";

export function OrderDetails() {
  // Input, output and errors come from the route.
  const order = useRoute(api.orders.getOrder, {
    input: { orderId: "order-1" },
    staleTime: 30_000,
  });

  return (
    <p>
      Order {order.data?.orderId}: {order.data?.status}
    </p>
  );
}
