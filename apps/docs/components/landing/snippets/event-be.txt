// A fact emitted after an order is created.
export default defineEvent({
  id: "orders.created",
  version: 1,

  input: z.object({
    orderId: z.string(),
    totalCents: z.number(),
  }),
});

// src/orders/events/order-confirmation.function.ts
// Subscribe independently, with durable delivery.
export const orderConfirmation = defineEventFunction({
  id: "orders.confirmation",
  event: "orders.created",
  delivery: "durable",

  handler: async (order, context) => {
    context.log.info("Order confirmation requested", {
      orderId: order.orderId,
    });
  },
});
