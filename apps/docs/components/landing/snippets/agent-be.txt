import lookupOrder from "lookup-order.tool.js";
import { createCommerceModel } from "commerce-model.js";

export default defineAgent({
  id: "orders.support",
  // LangChain runs the agent; RelKit checks its input and output.
  runtime: langchain({ model: createCommerceModel("support") }),
  input: z.object({ message: z.string() }),
  output: z.object({ answer: z.string() }),
  instructions: "Answer customer questions about their orders.",
  tools: [lookupOrder],
  client: { public: true },
  chat: { input: "message", output: "answer" },
  controls: ["stop"],
  limits: { maxSteps: 4, maxToolCalls: 4, timeoutMs: 60_000 },
});
