// src/services/auth/better-auth.ts — plain Better Auth.
export const auth = betterAuth({
  baseURL: process.env.BETTER_AUTH_URL,
  emailAndPassword: { enabled: true },
});

// src/services/auth/service.ts — RelKit mounts it at /api/auth.
export default defineBetterAuthService({ auth });

// src/routes/orders/[orderId]/route.ts
// Only signed-in customers can read an order.
export const GET = defineRoute({
  target: orders.getOrder,
  auth: "required",
});
