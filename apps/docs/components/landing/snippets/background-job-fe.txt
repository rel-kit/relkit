"use client";

export function ExportOrders() {
  // Start an order export; the receipt carries its run ID.
  const job = useJobTrigger("exportOrders");

  // Watch the accepted run for progress and completion.
  const { run } = useJobRun(
    "exportOrders",
    job.data ? { runId: job.data.runId } : {},
  );

  const start = () =>
    job.mutate({ input: { orderIds: ["order-1"] } });

  const label = run?.progress?.completed ?? run?.status;

  return (
    <button onClick={start} disabled={job.isPending}>
      {label ?? "Export orders"}
    </button>
  );
}
