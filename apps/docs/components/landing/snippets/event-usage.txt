export default defineFunction({
  input: z.object({
    orderId: z.string(),
    quantity: z.number().int().positive(),
  }),

  output: z.object({
    orderId: z.string(),
    totalCents: z.number(),
  }),

  dependencies: { events: { orderCreated } },

  handler: async ({ orderId, quantity }, context) => {
    const order = {
      orderId,
      totalCents: quantity * 1_000,
    };

    // Notify the order-confirmation subscriber.
    await context.events.orderCreated.publish(order);

    return order;
  },
});
