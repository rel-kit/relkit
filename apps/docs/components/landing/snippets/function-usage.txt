import createOrder from "create-order.function.js";

export default defineFunction({
  input: z.object({
    orderId: z.string(),
    sku: z.string(),
    quantity: z.number().int().positive(),
  }),

  output: z.object({
    orderId: z.string(),
    totalCents: z.number(),
  }),

  dependencies: { functions: { createOrder } },

  handler: async (input, context) => {
    // Call with validation, hooks, and tracing.
    const order = await context.functions.createOrder(input);

    return order;
  },
});
