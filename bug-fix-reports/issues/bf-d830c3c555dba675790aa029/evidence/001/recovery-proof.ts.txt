async function wait(label: string, fn: () => Promise<boolean>, ms = 60000) {
  const end = Date.now() + ms;
  while (Date.now() < end) {
    if (await fn()) return;
    await Bun.sleep(350);
  }
  throw new Error("Timed out: " + label);
}
async function headroom() {
  const end = Date.now() + 120000;
  const constrained = () => {
    const pressure = Number(
      execFileSync(
        "rtk",
        ["sysctl", "-n", "kern.memorystatus_vm_pressure_level"],
        { encoding: "utf8" },
      ).trim(),
    );
    const memory = execFileSync("rtk", ["memory_pressure", "-Q"], {
      encoding: "utf8",
    }).match(/free percentage: (\d+)%/);
    return pressure >= 4 || !memory || Number(memory[1]) < 40;
  };
  while (constrained()) {
    if (Date.now() > end)
      throw new Error(
        "Memory pressure persisted; experiment deferred with inputs restored",
      );
    await Bun.sleep(2000);
  }
}
async function serving(expected: string) {
  await wait("activation " + expected, async () => (await read()) === expected);
}

}
async function rejected(
  name: string,
  invalid: string,
  repaired: string,
  label: string,
) {
  await headroom();
  const previous = await read(),
    start = (await log()).length;
  await write(name, invalid);
  await wait(
    label + " diagnostic",
    async () =>
      (await log()).slice(start).includes("candidate.compile.failed") ||
      (await log()).slice(start).includes("candidate.spawn.failed"),
  );
  for (let n = 0; n < 3; n++)
    assert.equal(
      await read(),
      previous,
      "previous valid catalog must serve during rejection",
    );
  await write(name, repaired);
  await serving(previous);
  await wait(label + " repaired generation activation", async () =>
    (await log()).slice(start).includes('"state":"switch-succeeded"'),
  );
  passed(label + ": diagnostic, previous traffic, repaired activation");
}
